from aws_cdk import (
    Stack,
    Aws,
    Duration,
    CfnOutput,
    aws_lambda as _lambda,
    aws_dynamodb as dynamodb,
    aws_iam as iam,
    aws_s3 as s3,
    aws_sqs as sqs,
    aws_lambda_event_sources as lambda_events,
    aws_apigatewayv2 as apigwv2,
    aws_apigatewayv2_integrations as integrations,
    aws_apigatewayv2_authorizers as authorizers,
)
from constructs import Construct
import config


def name(base: str, stage: str) -> str:
    return f"{base}-{stage}"


class ArkaBackendStack(Stack):
    def __init__(self, scope: Construct, construct_id: str, stage: str, **kwargs):
        super().__init__(scope, construct_id, **kwargs)

        resource_stage = "dev" if stage == "dev-v2" else stage

        # ── SQS Queues ────────────────────────────────────────────────────────
        ai_analysis_dlq = sqs.Queue(
            self,
            "AIAnalysisDLQ",
            queue_name=name("ai-analysis-dlq", stage),
            retention_period=Duration.days(14),
        )

        ai_analysis_queue = sqs.Queue(
            self,
            "AIAnalysisQueue",
            queue_name=name(config.AI_ANALYSIS_QUEUE_NAME, stage),
            visibility_timeout=Duration.seconds(900),
            retention_period=Duration.days(4),
            dead_letter_queue=sqs.DeadLetterQueue(
                max_receive_count=3,
                queue=ai_analysis_dlq,
            ),
        )

        # ── IAM Role ──────────────────────────────────────────────────────────
        if stage == "dev-v2":
            role = iam.Role(
                self,
                "LambdaRole",
                role_name=config.INTERNAL_ADMIN_ROLE_NAME_DEV_V2,
                assumed_by=iam.ServicePrincipal("lambda.amazonaws.com"),
                managed_policies=[
                    iam.ManagedPolicy.from_aws_managed_policy_name(
                        "service-role/AWSLambdaBasicExecutionRole"
                    ),
                    iam.ManagedPolicy.from_aws_managed_policy_name(
                        "AmazonDynamoDBFullAccess"
                    ),
                    iam.ManagedPolicy.from_aws_managed_policy_name(
                        "AmazonS3FullAccess"
                    ),
                    iam.ManagedPolicy.from_aws_managed_policy_name(
                        "AmazonSSMReadOnlyAccess"
                    ),
                    iam.ManagedPolicy.from_aws_managed_policy_name(
                        "AmazonSQSFullAccess"
                    ),
                ],
            )
            cognito_user_pool_id = config.COGNITO_USER_POOL_ID_DEV_V2
            cognito_client_id = config.COGNITO_CLIENT_ID_DEV_V2
        else:
            role = iam.Role.from_role_name(
                self,
                "LambdaRole",
                name(config.LAMBDA_ROLE_NAME, stage),
            )
            cognito_user_pool_id = config.COGNITO_USER_POOL_ID
            cognito_client_id = config.COGNITO_CLIENT_ID

        # ── API Lambda (FastAPI, Docker image) ────────────────────────────────
        api_lambda = _lambda.DockerImageFunction(
            self,
            "FastApiLambda",
            function_name=name(config.LAMBDA_FUNCTION, stage),
            role=role,
            timeout=Duration.seconds(config.LAMBDA_TIMEOUT),
            memory_size=config.LAMBDA_MEMORY_SIZE,
            code=_lambda.DockerImageCode.from_image_asset(
                "../backend-development-KHM",
            ),
            environment={
                "STAGE": stage,
                "DYNAMODB_TABLE": name(config.DYNAMODB_TABLE_NAME, resource_stage),
                "S3_MEDIA_BUCKET": f"{config.S3_BUCKET_NAME}-{resource_stage}-{Aws.ACCOUNT_ID}",
                "COGNITO_REGION": config.COGNITO_REGION,
                "COGNITO_USER_POOL_ID": cognito_user_pool_id,
                "COGNITO_CLIENT_ID": cognito_client_id,
                "COOKIE_DOMAIN": "",  # host-only: bind cookies to this stage's API host, isolating dev/qa/prod
                "COOKIE_SAMESITE": "none" if stage == "dev-v2" else "lax",
                "AI_ANALYSIS_QUEUE_URL": ai_analysis_queue.queue_url,
                "AI_ANALYSIS_QUEUE_NAME": name(config.AI_ANALYSIS_QUEUE_NAME, stage),
                "OPENAI_API_KEY": config.OPENAI_API_KEY,
            },
        )

        ai_analysis_queue.grant_send_messages(api_lambda)

        # ── Worker Lambda (SQS consumer, Docker image) ────────────────────────
        worker_lambda = _lambda.DockerImageFunction(
            self,
            "AIAnalysisWorker",
            function_name=name("ai-analysis-worker", stage),
            role=role,
            timeout=Duration.seconds(900),
            memory_size=2048,
            code=_lambda.DockerImageCode.from_image_asset(
                "../backend-development-KHM",
                cmd=["lambda-workers.process_ai_analysis.handler"],
            ),
            environment={
                "STAGE": stage,
                "DYNAMODB_TABLE": name(config.DYNAMODB_TABLE_NAME, resource_stage),
                "S3_MEDIA_BUCKET": f"{config.S3_BUCKET_NAME}-{resource_stage}-{Aws.ACCOUNT_ID}",
                "AI_ANALYSIS_QUEUE_NAME": name(config.AI_ANALYSIS_QUEUE_NAME, stage),
                "OPENAI_API_KEY": config.OPENAI_API_KEY,
                "AWS_ACCOUNT_ID": Aws.ACCOUNT_ID,
            },
        )

        ai_analysis_queue.grant_consume_messages(worker_lambda)

        worker_lambda.add_event_source(
            lambda_events.SqsEventSource(
                ai_analysis_queue,
                batch_size=1,
                max_concurrency=5,
                report_batch_item_failures=True,
            )
        )

        # ── API Gateway ───────────────────────────────────────────────────────
        lambda_integration = integrations.HttpLambdaIntegration(
            "LambdaIntegration",
            api_lambda,
        )

        api = apigwv2.HttpApi(
            self,
            "HttpApi",
            api_name=name(config.API_GATEWAY, stage),
        )

        api.add_routes(
            path="/{proxy+}",
            methods=[apigwv2.HttpMethod.OPTIONS],
            integration=lambda_integration,
        )
        api.add_routes(
            path="/{proxy+}",
            methods=[apigwv2.HttpMethod.ANY],
            integration=lambda_integration,
        )

        for public_path in config.PUBLIC_PATHS:
            api.add_routes(
                path=public_path,
                methods=[apigwv2.HttpMethod.POST],
                integration=lambda_integration,
            )

        api.add_routes(
            path="/",
            methods=[apigwv2.HttpMethod.GET],
            integration=lambda_integration,
        )

        # ── Outputs ───────────────────────────────────────────────────────────
        CfnOutput(self, "ApiUrl", value=api.url)
        CfnOutput(self, "QueueUrl", value=ai_analysis_queue.queue_url)
        CfnOutput(self, "DLQUrl", value=ai_analysis_dlq.queue_url)
        CfnOutput(self, "QueueName", value=ai_analysis_queue.queue_name)
        CfnOutput(self, "ApiLambdaName", value=api_lambda.function_name)
        CfnOutput(self, "WorkerLambdaName", value=worker_lambda.function_name)
