function getCategoryClass(category?: string) {
  const key = category?.toLowerCase().split(/[ /_-]/)[0] || "general";
  return categoryConfig[key] || categoryConfig.general;
}

function getRaisedBy(defect: DefectData) {
  return defect.raised_by_crew?.name || defect.raised_by_inspector?.name || "Inspector";
}

function getCriticality(defect: DefectData) {
  const key = (defect.equipment_criticality || defect.priority || "medium").toLowerCase();
  return criticalityConfig[key] || criticalityConfig.medium;
}

function cn(...classes: Array<string | false | null | undefined>) {
  return classes.filter(Boolean).join(" ");
}

function formatDate(value?: string) {
  if (!value) return "-";
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return "-";
  return date.toLocaleDateString("en-GB", {
    day: "2-digit",
    month: "short",
    year: "numeric",
  });
}

function getDaysBetween(start: string, end: string) {
  const startDate = new Date(start);
  const endDate = new Date(end);
  if (Number.isNaN(startDate.getTime()) || Number.isNaN(endDate.getTime())) return 0;
  return Math.ceil(Math.abs(endDate.getTime() - startDate.getTime()) / (1000 * 60 * 60 * 24));
}

function getThirdPartyLabel(value: string) {
  return THIRD_PARTY_TYPES.find((type) => type.value === value)?.label ?? value.replace(/_/g, " ").toUpperCase();
}

function getEffectiveStatus(defect: DefectData) {
  const status = (defect.status ?? "open").toLowerCase();

  if (status === "closed" || status === "resolved") return "closed";

  if (defect.due_date) {
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const due = new Date(defect.due_date);
    due.setHours(0, 0, 0, 0);
    if (due < today) return "overdue";
  }

  if (status === "in_progress" || status === "progress") return "in_progress";

  return "open";
}

function canCloseDefect(defect: DefectData) {
  const status = (defect.status ?? "").toLowerCase();
  return status !== "closed" && status !== "resolved";
}

function Pagination({
  page,
  hasNext,
  total,
  limit,
  onPageChange,
  isLoading,
}: {
  page: number;
  hasNext: boolean;
  total: number;
  limit: number;
  onPageChange: (nextPage: number) => void;
  isLoading?: boolean;
}) {
  if (!total) return null;

  const start = (page - 1) * limit + 1;
  const end = Math.min(page * limit, total);

  return (
    <div className="flex items-center justify-between gap-3 border-t border-slate-100 bg-slate-50/60 px-5 py-3.5">
      <div className="text-[10px] font-bold uppercase tracking-[0.2em] text-slate-400">
        Showing {start}-{end} of {total}
      </div>
      <div className="flex items-center gap-2">
        <button
          onClick={() => onPageChange(page - 1)}
          disabled={page === 1 || isLoading}
          className={cn(
            "rounded-md border px-3 py-1.5 text-xs font-semibold transition-colors",
            page === 1 || isLoading
              ? "cursor-not-allowed border-slate-200 bg-slate-100 text-slate-400"
              : "border-slate-300 bg-white text-slate-600 hover:bg-slate-50"
          )}
        >
          Previous
        </button>
        <span className="px-1 text-xs font-semibold text-slate-600">{page}</span>
        <button
          onClick={() => onPageChange(page + 1)}
          disabled={!hasNext || isLoading}
          className={cn(
            "rounded-md border px-3 py-1.5 text-xs font-semibold transition-colors",
            !hasNext || isLoading
              ? "cursor-not-allowed border-slate-200 bg-slate-100 text-slate-400"
              : "border-slate-300 bg-white text-slate-600 hover:bg-slate-50"
          )}
        >
          Next
        </button>
      </div>
    </div>
  );
}

function FindingRow({
  finding,
  onSelect,
  onCloseDefect,
  onDeleteDefect,
}: {
  finding: DefectData;
  onSelect: () => void;
  onCloseDefect: (finding: DefectData) => void;
  onDeleteDefect: (finding: DefectData) => void;
}) {
  const effectiveStatus = getEffectiveStatus(finding);
  const status = statusConfig[effectiveStatus] ?? statusConfig.open;
  const criticality = getCriticality(finding);
  const isClosed = effectiveStatus === "closed";
  const isOverdue = effectiveStatus === "overdue";
  const age = finding.created_at ? getDaysBetween(finding.created_at, new Date().toISOString()) : 0;
  const title = finding.title || finding.description || "Untitled finding";
  const location = finding.location_on_ship || finding.equipment_name || "General area";
  const category = finding.category || finding.event_type || getThirdPartyLabel(finding.inspection_type || "") || "Deficiency";
  const raisedBy = getRaisedBy(finding);

  return (
    <tr
      onClick={onSelect}
      className="cursor-pointer border-b border-slate-100 transition-colors hover:bg-slate-50/70"
    >
      <td className="pl-5 pr-3 py-3 align-middle">
        <div className="font-mono text-[12px] font-bold tracking-tight text-slate-900">
          {finding.reference_number || finding.defect_id?.slice(-8) || "N/A"}
        </div>
      </td>
      <td className={tableCellClass}>
        <div className="max-w-[220px] truncate text-[13px] font-semibold text-slate-800" title={title}>
          {title}
        </div>
      </td>
      <td className={tableCellClass}>
        <div className="max-w-[160px] truncate" title={location}>{location}</div>
      </td>
      <td className="px-3 py-3 align-middle whitespace-nowrap">
        <span className={cn(pillClass, status.className)}>{status.label}</span>
      </td>
      <td className="px-3 py-3 align-middle">
        <span className={cn(pillClass, getCategoryClass(category))}>
          <span className="max-w-[120px] truncate" title={category}>{category}</span>
        </span>
      </td>
      <td className="px-3 py-3 align-middle whitespace-nowrap">
        <span className={cn(pillClass, criticality.className)} title={criticality.label}>
          <span className={cn("h-2 w-2 rounded-full", criticality.dot)} />
          <span className="hidden xl:inline">{criticality.label}</span>
        </span>
      </td>
      <td className={tableCellClass}>
        <div className="max-w-[140px] truncate" title={raisedBy}>{raisedBy}</div>
      </td>
      <td className="px-3 py-3 align-middle whitespace-nowrap text-[13px] font-medium text-slate-500">{formatDate(finding.created_at)}</td>
      <td className="px-3 py-3 align-middle whitespace-nowrap">
        {finding.due_date ? (
          <span className={cn("inline-flex items-center gap-1 text-[13px] font-medium", isOverdue && !isClosed ? "text-red-600" : "text-slate-500")}>
            <Calendar className="h-3.5 w-3.5" />
            {formatDate(finding.due_date)}
          </span>
        ) : (
          <span className="text-[13px] text-slate-400">-</span>
        )}
      </td>
      <td className="px-3 py-3 align-middle whitespace-nowrap text-[13px] font-medium text-slate-500">
        {formatDate(finding.updated_at || finding.closed_at || "")}
      </td>
      <td className="px-3 py-3 align-middle">
        <div className="flex items-center gap-2">
          {canCloseDefect(finding) ? (
            <button
              type="button"
              onClick={(event) => {
                event.stopPropagation();
                onCloseDefect(finding);
              }}
              className="inline-flex items-center gap-1 rounded-md border border-emerald-200 bg-emerald-50 px-2.5 py-1.5 text-xs font-bold text-emerald-700 transition-colors hover:bg-emerald-100"
            >
              <CheckCircle className="h-3.5 w-3.5" />
              Close
            </button>
          ) : null}
          <button
            type="button"
            onClick={(event) => {
              event.stopPropagation();
              onDeleteDefect(finding);
            }}
            className="inline-flex items-center gap-1 rounded-md border border-red-200 bg-red-50 px-2.5 py-1.5 text-xs font-bold text-red-700 transition-colors hover:bg-red-100"
          >
            <Trash2 className="h-3.5 w-3.5" />
            Delete
          </button>
        </div>
      </td>
      <td className="whitespace-nowrap pl-3 pr-5 py-3 text-right">
        <div className="flex flex-col items-end leading-none">
          <span className={cn("text-sm font-bold", isOverdue ? "text-red-600" : age > 40 ? "text-amber-600" : "text-slate-700")}>
            {isClosed ? "-" : age}
          </span>
        </div>
      </td>
    </tr>
  );
}
function FindingsTable({
  vesselId,
  vesselName,
  selectedInspectionType,
  onInspectionTypeChange,
  findings,
  isLoading,
  page,
  onPageChange,
  onSelectFinding,
  onCloseDefect,
  onDeleteDefect,
}: {
  vesselId: string;
  vesselName: string;
  selectedInspectionType: string;
  onInspectionTypeChange: (value: string) => void;
  findings: DefectData[];
  isLoading: boolean;
  page: number;
  onPageChange: (page: number) => void;
  onSelectFinding: (finding: DefectData) => void;
  onCloseDefect: (finding: DefectData) => void;
  onDeleteDefect: (finding: DefectData) => void;
}) {
  const [searchTerm, setSearchTerm] = useState("");
  const [statusFilter, setStatusFilter] = useState("all");
  const [categoryFilter, setCategoryFilter] = useState("all");

  const categoryOptions = useMemo(() => {
    return Array.from(new Set(findings.map((finding) => finding.category || finding.event_type || "Deficiency"))).filter(Boolean);
  }, [findings]);

  const filteredFindings = useMemo(() => {
    return findings.filter((finding) => {
      const effectiveStatus = getEffectiveStatus(finding);
      const category = finding.category || finding.event_type || "Deficiency";
      const matchesSearch =
        !searchTerm ||
        [finding.reference_number, finding.location_on_ship, finding.title, finding.description]
          .filter(Boolean)
          .join(" ")
          .toLowerCase()
          .includes(searchTerm.toLowerCase());
      const matchesStatus = statusFilter === "all" || effectiveStatus === statusFilter;
      const matchesCategory = categoryFilter === "all" || category === categoryFilter;
      return matchesSearch && matchesStatus && matchesCategory;
    });
  }, [categoryFilter, findings, searchTerm, statusFilter]);

  const paginatedFindings = filteredFindings.slice((page - 1) * FINDINGS_PER_PAGE, page * FINDINGS_PER_PAGE);
  const totalPages = Math.ceil(filteredFindings.length / FINDINGS_PER_PAGE);
  const hasMore = page < totalPages;

  return (
    <div className="overflow-hidden rounded-xl border border-slate-200 bg-white shadow-sm">
      <div className="flex flex-col gap-3 border-b border-slate-200 bg-slate-50/70 px-4 py-3 lg:flex-row lg:items-center lg:justify-between">
        <div>
          <div className="flex items-center gap-2">
            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-sky-50">
              <ClipboardCheck className="h-4 w-4 text-sky-600" />
            </div>
            <div>
              <h3 className="text-sm font-extrabold text-slate-900">{vesselName}</h3>
              <p className="text-[10px] font-medium text-slate-400">Third party inspection findings</p>
            </div>
          </div>
          <p className="mt-2 text-[11px] text-slate-500">
            Viewing <span className="font-semibold text-slate-700">{getThirdPartyLabel(selectedInspectionType)}</span>
          </p>
        </div>
         <div className="flex items-center gap-3">
          <DownloadButton
              config={{
                filterType: "inspection_type",
                filterValue: selectedInspectionType,
                vesselId: vesselId,
                vesselName: vesselName,
                title: `${getThirdPartyLabel(selectedInspectionType)} Report`,
              }}
              variant="outline"
              size="sm"
            />
        <div className="relative w-full lg:w-[260px]">
          <select
            value={selectedInspectionType}
            onChange={(e) => onInspectionTypeChange(e.target.value)}
            className="w-full appearance-none rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-9 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
          >
            {THIRD_PARTY_TYPES.map((type) => (
              <option key={type.value} value={type.value}>
                {type.label}
              </option>
            ))}
          </select>
          <Filter className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
          <ChevronDown className="pointer-events-none absolute right-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
        </div>
      </div>
      </div>

      <div className="border-b border-slate-200 bg-white px-4 py-3">
        <div className="flex flex-wrap items-center gap-3">
          <div className="relative w-full max-w-xs">
            <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
            <input
              value={searchTerm}
              onChange={(e) => setSearchTerm(e.target.value)}
              placeholder="Search finding..."
              className="w-full rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-4 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
            />
          </div>

          <div className="relative">
            <Filter className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
            <select
              value={statusFilter}
              onChange={(e) => setStatusFilter(e.target.value)}
              className="appearance-none rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-9 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
            >
              <option value="all">All statuses</option>
              <option value="open">Open</option>
              <option value="in_progress">In Progress</option>
              <option value="overdue">Overdue</option>
              <option value="closed">Closed</option>
            </select>
            <ChevronDown className="pointer-events-none absolute right-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
          </div>

          <div className="relative">
            <Ship className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
            <select
              value={categoryFilter}
              onChange={(e) => setCategoryFilter(e.target.value)}
              className="appearance-none rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-9 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
            >
              <option value="all">All categories</option>
              {categoryOptions.map((option) => (
                <option key={option} value={option}>
                  {option}
                </option>
              ))}
            </select>
            <ChevronDown className="pointer-events-none absolute right-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
          </div>
        </div>
      </div>

      {isLoading ? (
        <div className="p-4">
          <TableSkeleton rows={6} />
        </div>
      ) : paginatedFindings.length === 0 ? (
        <div className="flex flex-col items-center justify-center py-16 text-center">
          <ClipboardCheck className="mb-3 h-12 w-12 text-slate-300" />
          <p className="text-sm font-medium text-slate-400">
            No findings found for {getThirdPartyLabel(selectedInspectionType)} on {vesselName}.
          </p>
        </div>
      ) : (
        <div className="overflow-x-auto">
          <table className="min-w-[1440px] w-full table-fixed border-collapse text-left">
            <thead className="border-b border-slate-200 bg-slate-50/70">
              <tr>
                <th className={cn(tableHeadClass, "w-[150px] pl-5")}>Defect Number / Reference Number</th>
                <th className={cn(tableHeadClass, "w-[240px]")}>Title</th>
                <th className={cn(tableHeadClass, "w-[170px]")}>Location</th>
                <th className={cn(tableHeadClass, "w-[120px]")}>Status</th>
                <th className={cn(tableHeadClass, "w-[140px]")}>Category</th>
                <th className={cn(tableHeadClass, "w-[120px]")}>Criticality</th>
                <th className={cn(tableHeadClass, "w-[150px]")}>Raised By</th>
                <th className={cn(tableHeadClass, "w-[120px]")}>Created Date</th>
                <th className={cn(tableHeadClass, "w-[120px]")}>Due Date</th>
                <th className={cn(tableHeadClass, "w-[120px]")}>Updated Date</th>
                <th className={cn(tableHeadClass, "w-[170px]")}>Action</th>
                <th className={cn(tableHeadClass, "w-[90px] pr-5 text-right")}>Age</th>
              </tr>
            </thead>
            <tbody>
              {paginatedFindings.map((finding, index) => (
                <FindingRow
                  key={finding.defect_id || `${vesselId}-${index}`}
                  finding={finding}
                  onSelect={() => onSelectFinding(finding)}
                  onCloseDefect={onCloseDefect}
                  onDeleteDefect={onDeleteDefect}
                />
              ))}
            </tbody>
          </table>
        </div>
      )}

      <Pagination
        page={page}
        hasNext={hasMore}
        total={filteredFindings.length}
        limit={FINDINGS_PER_PAGE}
        onPageChange={onPageChange}
        isLoading={isLoading}
      />
    </div>
  );
}

export default function ThirdPartyPage() {
  const [fleetStats, setFleetStats] = useState<FleetStats | null>(null);
  const [vessels, setVessels] = useState<VesselStats[]>([]);
  const [isLoadingVessels, setIsLoadingVessels] = useState(true);
  const [isLoadingStats, setIsLoadingStats] = useState(true);
  const [searchTerm, setSearchTerm] = useState("");
  const [typeFilter, setTypeFilter] = useState("all");

  const [activeVesselId, setActiveVesselId] = useState<string | null>(null);
  const [activeVesselName, setActiveVesselName] = useState("");
  const [activeVesselType, setActiveVesselType] = useState("");

  const [detailFindings, setDetailFindings] = useState<DefectData[]>([]);
  const [detailPage, setDetailPage] = useState(1);
  const [, setDetailHasNext] = useState(false);
  const [, setDetailTotal] = useState(0);
  const [detailInspectionType, setDetailInspectionType] = useState(DEFAULT_INSPECTION_TYPE);
  const [isLoadingDetail, setIsLoadingDetail] = useState(false);
  const [selectedDefect, setSelectedDefect] = useState<DefectData | null>(null);
  const [activeActionDefect, setActiveActionDefect] = useState<DefectData | null>(null);
  const [activeAction, setActiveAction] = useState<"close" | "delete" | null>(null);
  const [closingRemarks, setClosingRemarks] = useState("");
  const [isActionSubmitting, setIsActionSubmitting] = useState(false);
  const [toast, setToast] = useState<ActionToastState>(null);
  const searchParams = useSearchParams();
  const fetchFleetStatistics = useCallback(async () => {
    setIsLoadingStats(true);
    try {
      const response = await defectsApi.getFleetStatistics();
      if (response.success) {
        setFleetStats(response.data);
      }
    } catch (error) {
      console.error("Failed to fetch fleet statistics:", error);
    } finally {
      setIsLoadingStats(false);
    }
  }, []);

  // Fetch vessels - USING BACKEND DATA DIRECTLY (NO EXTRA API CALLS)
  const fetchVessels = useCallback(async () => {
    setIsLoadingVessels(true);
    try {
      const response = await defectsApi.getVesselsWithStats("third_party");
      if (!response.success || !response.data?.vessels) return;

      const mapped: VesselStats[] = response.data.vessels.map((vessel) => {
        const thirdPartyStats = vessel.third_party_stats || {};
        
        const inspection_counts: Partial<Record<string, number>> = {};
        const inspection_open_counts: Partial<Record<string, number>> = {};
        
        // Use data from backend - NO extra API calls!
        for (const type of THIRD_PARTY_TYPES) {
          const stats = thirdPartyStats[type.value] || {};
          inspection_counts[type.value] = stats.total || 0;
          inspection_open_counts[type.value] = stats.open || 0;
        }

        return {
          vessel_id: vessel.vessel_id,
          name: vessel.name,
          type: vessel.type,
          imo_number: vessel.imo_number,
          total_defects: vessel.total_defects,
          open: vessel.open,
          overdue: vessel.overdue,
          closed: vessel.closed,
          closed_percentage: vessel.closed_percentage,
          inspection_counts,
          inspection_open_counts,
        };
      });

      setVessels(mapped);
    } catch (error) {
      console.error("Failed to fetch vessels:", error);
    } finally {
      setIsLoadingVessels(false);
    }
  }, []);
    useEffect(() => {
    const vesselId = searchParams.get("vesselId");
    const page = searchParams.get("page");
    const type = searchParams.get("type");
    
    if (vesselId && vessels.length > 0) {
      const vessel = vessels.find(v => v.vessel_id === vesselId);
      if (vessel) {
        setActiveVesselId(vessel.vessel_id);
        setActiveVesselName(vessel.name);
        setActiveVesselType(vessel.type);
        if (type) setDetailInspectionType(type);
        if (page) setDetailPage(parseInt(page));
      }
    }
  }, [searchParams, vessels]);
  const fetchDetailFindings = useCallback(async (vesselId: string, inspectionType: string, page: number) => {
    setIsLoadingDetail(true);
    try {
      const response = await defectsApi.getAll(page, FINDINGS_PER_PAGE, inspectionType, undefined, vesselId);

      if (!response.success) {
        setDetailFindings([]);
        setDetailHasNext(false);
        setDetailTotal(0);
        return;
      }

      const items: DefectData[] = response.data?.items ?? [];
      setDetailFindings(items);
      setDetailHasNext(Boolean(response.data?.has_next));
      const payload = response.data as typeof response.data & { total?: number };
      setDetailTotal(payload.total ?? items.length + (response.data?.has_next ? 1 : 0));
    } catch (error) {
      console.error("Failed to fetch detail findings:", error);
      setDetailFindings([]);
      setDetailHasNext(false);
      setDetailTotal(0);
    } finally {
      setIsLoadingDetail(false);
    }
  }, []);

  useEffect(() => {
    fetchFleetStatistics();
    fetchVessels();
  }, [fetchFleetStatistics, fetchVessels]);

  useEffect(() => {
    if (!activeVesselId) return;
    fetchDetailFindings(activeVesselId, detailInspectionType, detailPage);
  }, [activeVesselId, detailInspectionType, detailPage, fetchDetailFindings]);

  const fleetKpis = useMemo(() => {
    const thirdPartyStats = fleetStats?.third_party ?? {};
    type FleetKpiSummary = {
      vessels: number;
      total: number;
      open: number;
      overdue: number;
      closed: number;
    };

    return Object.values(thirdPartyStats).reduce<FleetKpiSummary>(
      (acc, stats) => {
        acc.total += stats.total ?? 0;
        acc.open += stats.open ?? 0;
        acc.overdue += stats.overdue ?? 0;
        acc.closed += stats.closed ?? 0;
        return acc;
      },
      {
        vessels: fleetStats?.vessels_count ?? vessels.length,
        total: 0,
        open: 0,
        overdue: 0,
        closed: 0,
      }
    );
  }, [fleetStats, vessels.length]);

  const vesselTypes = useMemo(() => Array.from(new Set(vessels.map((vessel) => vessel.type))).filter(Boolean), [vessels]);

  const filteredVessels = useMemo(() => {
    return vessels.filter((vessel) => {
      const matchesName = vessel.name.toLowerCase().includes(searchTerm.toLowerCase());
      const matchesType = typeFilter === "all" || vessel.type === typeFilter;
      return matchesName && matchesType;
    });
  }, [searchTerm, typeFilter, vessels]);

  const handleOpenVessel = (vessel: VesselStats) => {
    setActiveVesselId(vessel.vessel_id);
    setActiveVesselName(vessel.name);
    setActiveVesselType(vessel.type);
    setDetailPage(1);
    setDetailInspectionType(DEFAULT_INSPECTION_TYPE);
    setDetailFindings([]);
    setDetailHasNext(false);
    setDetailTotal(0);
    setSelectedDefect(null);
  };

  const handleBack = () => {
    setActiveVesselId(null);
    setActiveVesselName("");
    setActiveVesselType("");
    setDetailFindings([]);
    setDetailPage(1);
    setDetailHasNext(false);
    setDetailTotal(0);
    setDetailInspectionType(DEFAULT_INSPECTION_TYPE);
    setSelectedDefect(null);
  };

  const handleSelectFinding = (finding: DefectData) => {
    sessionStorage.setItem(
      "defectNavigationContext",
      JSON.stringify({
        sourcePage: "third-party",
        vesselId: activeVesselId,
        currentPage: detailPage,
        filters: {
          inspectionType: detailInspectionType,
          status: getEffectiveStatus(finding),
        },
        defectIds: detailFindings.map((item) => item.defect_id),
        currentIndex: detailFindings.findIndex((item) => item.defect_id === finding.defect_id),
        timestamp: Date.now(),
      }),
    );

    setSelectedDefect(finding);
  };

  useEffect(() => {
    if (!toast) return;
    const timeoutId = window.setTimeout(() => setToast(null), 3000);
    return () => window.clearTimeout(timeoutId);
  }, [toast]);

  const closeActionDialog = () => {
    if (isActionSubmitting) return;
    setActiveAction(null);
    setActiveActionDefect(null);
    setClosingRemarks("");
  };

  const handleOpenCloseDialog = (finding: DefectData) => {
    setActiveAction("close");
    setActiveActionDefect(finding);
    setClosingRemarks("");
  };

  const handleOpenDeleteDialog = (finding: DefectData) => {
    setActiveAction("delete");
    setActiveActionDefect(finding);
    setClosingRemarks("");
  };

  const handleConfirmAction = async () => {
    if (!activeAction || !activeActionDefect) return;

    const vesselId = activeActionDefect.vessel_id || activeVesselId;
    if (!vesselId) {
      setToast({ type: "error", message: "Unable to identify the vessel for this defect" });
      return;
    }

    setIsActionSubmitting(true);
    try {
      if (activeAction === "close") {
        const response = await defectsApi.closeDefect(vesselId, activeActionDefect.defect_id);
        if (!response.success) throw new Error(response.message || "Failed to close defect");
      } else {
        const response = await defectsApi.delete(vesselId, activeActionDefect.defect_id);
        if (!response.success) throw new Error(response.message || "Failed to delete defect");
      }

      if (activeAction === "delete") {
        setSelectedDefect((current) => (current?.defect_id === activeActionDefect.defect_id ? null : current));
      }

      setActiveAction(null);
      setActiveActionDefect(null);
      setClosingRemarks("");

      if (activeVesselId) {
        await fetchDetailFindings(activeVesselId, detailInspectionType, detailPage);
      }
      await fetchFleetStatistics();
      await fetchVessels();
      setToast({
        type: "success",
        message: activeAction === "close" ? "Defect closed successfully" : "Defect deleted successfully",
      });
    } catch (error) {
      console.error(`Failed to ${activeAction} defect:`, error);
      setToast({
        type: "error",
        message: error instanceof Error ? error.message : `Failed to ${activeAction} defect`,
      });
    } finally {
      setIsActionSubmitting(false);
    }
  };

  if ((isLoadingVessels || isLoadingStats) && !activeVesselId) {
    return (
      <div className="mx-auto min-h-screen max-w-[1600px] bg-slate-50 p-6 text-slate-800">
        <div className="mb-6 flex flex-col justify-between gap-3 md:flex-row md:items-end">
          <div className="space-y-2">
            <SkeletonBlock className="h-6 w-64" />
            <SkeletonBlock className="h-4 w-72" />
          </div>
        </div>

        <div className="mb-6 grid grid-cols-2 gap-6 md:grid-cols-5">
          {Array.from({ length: 5 }).map((_, index) => (
            <div key={index} className="rounded-lg border border-gray-100 bg-white p-4 shadow-sm">
              <div className="flex items-start justify-between gap-3">
                <div className="space-y-2">
                  <SkeletonBlock className="h-3 w-20" />
                  <SkeletonBlock className="h-8 w-16" />
                  <SkeletonBlock className="h-3 w-14" />
                </div>
                <SkeletonBlock className="h-4 w-4" />
              </div>
            </div>
          ))}
        </div>

        <div className="mb-6 flex flex-wrap items-center gap-6">
          <SkeletonBlock className="h-10 w-full max-w-xs" />
          <SkeletonBlock className="h-10 w-40" />
        </div>

        <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
          {Array.from({ length: 6 }).map((_, index) => (
            <div key={index} className="overflow-hidden rounded-xl border border-slate-100 bg-white shadow-sm">
              <div className="p-4">
                <div className="mb-3 flex items-start justify-between gap-3">
                  <div className="flex min-w-0 items-start gap-3">
                    <SkeletonBlock className="h-9 w-9 rounded-lg" />
                    <div className="min-w-0 space-y-2">
                      <SkeletonBlock className="h-4 w-36" />
                      <SkeletonBlock className="h-3 w-24" />
                    </div>
                  </div>
                  <SkeletonBlock className="h-5 w-5 rounded-full" />
                </div>
                <SkeletonBlock className="mb-3 h-5 w-24 rounded-full" />
                <div className="border-t border-slate-100 pt-3">
                  {Array.from({ length: 2 }).map((__, rowIndex) => (
                    <div key={rowIndex} className="flex items-center justify-between py-2">
                      <SkeletonBlock className="h-3 w-24" />
                      <SkeletonBlock className="h-3 w-16" />
                    </div>
                  ))}
                </div>
                <div className="mt-3 border-t border-slate-100 pt-2">
                  <SkeletonBlock className="h-4 w-20" />
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>
    );
  }

  return (
    <div className="mx-auto min-h-screen max-w-[1600px] bg-slate-50 p-6 text-slate-800">
      <div className="mb-6 flex flex-col justify-between gap-3 md:flex-row md:items-end">
        <div>
          <h1 className="text-xl font-black tracking-tight text-slate-900">Third Party Findings</h1>
          <p className="mt-0.5 text-xs font-medium text-slate-500">Track third party findings across the fleet</p>
        </div>
      </div>

      <div className="mb-6 grid grid-cols-2 gap-6 md:grid-cols-5">
        <KPIBlock label="Vessels" value={fleetKpis.vessels} subtext="in fleet" icon={<Ship className="h-4 w-4" />} highlightClassName="text-sky-600" />
        <KPIBlock label="Total findings" value={fleetKpis.total} subtext="all third party types" icon={<ClipboardCheck className="h-4 w-4" />} />
        <KPIBlock label="Open" value={fleetKpis.open} subtext="unresolved" icon={<AlertTriangle className="h-4 w-4" />} highlightClassName="text-amber-600" />
        <KPIBlock label="Overdue" value={fleetKpis.overdue} subtext="past due" icon={<AlertTriangle className="h-4 w-4" />} highlightClassName="text-red-600" />
        <KPIBlock label="Closed" value={fleetKpis.closed} subtext="resolved" icon={<CheckCircle className="h-4 w-4" />} highlightClassName="text-emerald-600" />
      </div>

      {!activeVesselId ? (
        <>
          <div className="mb-6 flex flex-wrap items-center gap-6">
            <div className="relative w-full max-w-xs">
              <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
              <input
                type="text"
                placeholder="Search vessel..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="w-full rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-4 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
              />
            </div>

            <div className="relative">
              <Ship className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
              <select
                value={typeFilter}
                onChange={(e) => setTypeFilter(e.target.value)}
                className="appearance-none rounded-lg border border-slate-200 bg-white py-2 pl-9 pr-9 text-sm font-medium text-slate-700 focus:outline-none focus:ring-2 focus:ring-sky-500"
              >
                <option value="all">All vessel types</option>
                {vesselTypes.map((type) => (
                  <option key={type} value={type}>
                    {type}
                  </option>
                ))}
              </select>
              <ChevronDown className="pointer-events-none absolute right-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" />
            </div>
          </div>

          {filteredVessels.length === 0 ? (
            <div className="rounded-xl border border-slate-200 bg-white p-16 text-center shadow-sm">
              <ClipboardCheck className="mx-auto mb-3 h-12 w-12 text-slate-300" />
              <p className="text-sm font-medium text-slate-400">No third party inspections match the current filters.</p>
            </div>
          ) : (
            <div className="grid grid-cols-1 gap-6 sm:grid-cols-2 lg:grid-cols-3">
              {filteredVessels.map((vessel) => (
                <VesselCard key={vessel.vessel_id} vessel={vessel} onClick={() => handleOpenVessel(vessel)} />
              ))}
            </div>
          )}
        </>
      ) : (
        <div className="space-y-4">
          <div className="rounded-xl border border-slate-200 bg-white px-4 py-3 shadow-sm">
            <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
              <div className="flex items-center gap-3">
                <button
                  type="button"
                  onClick={handleBack}
                  className="inline-flex items-center gap-1.5 rounded-md border border-slate-300 bg-white px-3 py-1.5 text-xs font-semibold text-slate-600 shadow-sm transition-colors hover:bg-slate-50"
                >
                  <ChevronLeft className="h-3.5 w-3.5" />
                  All vessels
                </button>
                <div className="hidden h-5 w-px bg-slate-200 sm:block" />
                <div className="min-w-0">
                  <h2 className="truncate text-sm font-extrabold text-slate-900">{activeVesselName}</h2>
                  <p className="text-[11px] font-medium text-slate-400">{activeVesselType}</p>
                </div>
              </div>
            </div>
          </div>

          <FindingsTable
            vesselId={activeVesselId}
            vesselName={activeVesselName}
            selectedInspectionType={detailInspectionType}
            onInspectionTypeChange={(value) => {
              setDetailInspectionType(value);
              setDetailPage(1);
              setSelectedDefect(null);
            }}
            findings={detailFindings}
            isLoading={isLoadingDetail}
            page={detailPage}
            onPageChange={(nextPage) => setDetailPage(nextPage)}
            onSelectFinding={handleSelectFinding}
            onCloseDefect={handleOpenCloseDialog}
            onDeleteDefect={handleOpenDeleteDialog}
          />
        </div>
      )}

      <DefectActionDialog
        action={activeAction || "close"}
        defect={activeActionDefect}
        isOpen={Boolean(activeAction && activeActionDefect)}
        isSubmitting={isActionSubmitting}
        onClose={closeActionDialog}
        onConfirm={handleConfirmAction}
      />

      <ActionToast toast={toast} onClose={() => setToast(null)} />

      {selectedDefect && (activeVesselId || selectedDefect.vessel_id) && (
        <DefectSummaryPanel
          defect={selectedDefect}
          vesselId={activeVesselId || selectedDefect.vessel_id || ""}
          onClose={() => setSelectedDefect(null)}
        />
      )}
    </div>
  );
}
