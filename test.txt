def lambda_download_response(result):
    """Convert Lambda-style binary download responses into FastAPI responses."""
    if isinstance(result, dict) and result.get("statusCode") == 302:
        location = result.get("headers", {}).get("Location")
        if location:
            return RedirectResponse(url=location)

    if isinstance(result, dict) and result.get("isBase64Encoded"):
        headers = result.get("headers", {})
        content_type = headers.get("Content-Type", "application/octet-stream")
        response_headers = {
            key: value
            for key, value in headers.items()
            if key.lower() not in {"content-type", "content-transfer-encoding"}
        }
        return Response(
            content=base64.b64decode(result["body"]),
            status_code=result.get("statusCode", 200),
            headers=response_headers,
            media_type=content_type,
        )

    if isinstance(result, dict) and "body" in result:
        headers = result.get("headers", {})
        content_type = headers.get("Content-Type", "application/json")
        return Response(
            content=result["body"],
            status_code=result.get("statusCode", 200),
            headers={key: value for key, value in headers.items() if key.lower() != "content-type"},
            media_type=content_type,
        )

    return result

@vessel_router.get(
    "/{vessel_id}/defects/pdf", summary="Download Vessel Defect Report PDF"
)
async def download_defect_pdf_route(
    vessel_id: str,
    request: Request,
    tenant_ctx=Depends(require_role("admin", "inspector")),
):
    event = await lambda_to_fastapi_response(
        request, path_params={"vessel_id": vessel_id}
    )
    result = download_defect_pdf_handler(event, None)

    # Branch 1: Handle HTTP Redirects (for large S3-backed PDFs)
    if isinstance(result, dict) and result.get("statusCode") == 302:
        location = result.get("headers", {}).get("Location")
        if location:
            return RedirectResponse(url=location)

    # Branch 2: Handle inline Binary/Base64 PDFs
    if isinstance(result, dict) and result.get("isBase64Encoded"):
        headers = result.get("headers", {})
        pdf_bytes = base64.b64decode(result["body"])

        # Ensure Content-Disposition is passed if present
        resp_headers = {k: v for k, v in headers.items() if k.lower() != "content-type"}

        return Response(
            content=pdf_bytes,
            status_code=result.get("statusCode", 200),
            headers=resp_headers,
            media_type="application/pdf",
        )

    # Branch 3: Handle structured JSON responses (for redirects or errors)
    if isinstance(result, dict) and "body" in result:
        headers = result.get("headers", {})
        content_type = headers.get("Content-Type", "application/json")

        return Response(
            content=result["body"],
            status_code=result.get("statusCode", 200),
            headers={k: v for k, v in headers.items() if k.lower() != "content-type"},
            media_type=content_type,
        )

    # Branch 4: Default pass-through
    return result

            content=base64.b64decode(result["body"]),
            media_type="application/pdf",
            headers={"Content-Disposition": f'attachment; filename="incident_report_{incident_id}.pdf"'}
        )
    
    return Response(
        content=result.get("body", "Internal Server Error"),
        status_code=result.get("statusCode", 500),
        media_type="application/json"
    )
